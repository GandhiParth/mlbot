conv_atr = ta.atr(conv_atr_period)

conv_n   = 0.0
conv_sum = 0.0

    conv_n   += 1
    conv_sum += close

    conv_n   += 1

if conv_use_ema21

    conv_sum += ema21

    conv_sum += sma50
conv_mean = conv_n > 0 ? conv_sum / conv_n : 0.0

conv_var = 0.0
if conv_use_close
    conv_var += (close - conv_mean) * (close - conv_mean)
if conv_use_ema9
    conv_var += (ema9 - conv_mean)  * (ema9 - conv_mean)
if conv_use_ema21
    conv_var += (ema21 - conv_mean) * (ema21 - conv_mean)
if conv_use_sma50
    conv_var += (sma50 - conv_mean) * (sma50 - conv_mean)

conv_pop_std = conv_n >= 2 ? math.sqrt(conv_var / conv_n) : 0.0
dispersion   = not na(conv_atr) and conv_atr > 0 and conv_n >= 2 ? conv_pop_std / conv_atr : na
disp_text    = not na(dispersion) ? str.tostring(math.round(dispersion, 3)) : "N/A"

// Rolling dispersion array (used by backward-scan)
var disp_arr = array.new_float(0)
array.push(disp_arr, dispersion)
if array.size(disp_arr) > conv_max_bars + 5
    array.shift(disp_arr)


// ============================================================
// MA CONVERGENCE: backward window scan
// ============================================================
// Original-style scan: for each window size w from conv_min_bars to conv_max_bars,
// average dispersion over the most recent w bars. Increase w as long as avg < cutoff.
// Stop at the first w where avg >= cutoff (or invalid). best_window = largest valid w.

calc_best_window() =>
    d_size = array.size(disp_arr)
    bw = 0
    if d_size >= conv_min_bars
        for w = conv_min_bars to math.min(conv_max_bars, d_size)
            d_sum = 0.0
            valid = true
            for j = 0 to w - 1
                idx = d_size - 1 - j
                if idx < 0
                    valid := false
                    break
                d_val = array.get(disp_arr, idx)
                if na(d_val)
                    valid := false
                    break
                d_sum += d_val
            if valid
                d_avg = d_sum / w
                if d_avg < conv_cutoff
                    bw := w
                else
                    break
            else
                break
    bw

best_window = calc_best_window()
is_in_conv  = best_window >= conv_min_bars


// ============================================================
// MA CONVERGENCE: state machine (single box per period)
// ============================================================

var bool  prev_in_conv  = false
var int   conv_start    = na
var float conv_zone_hi  = na
var float conv_zone_lo  = na
var float conv_disp_sum = 0.0
var int   conv_disp_cnt = 0

var c_start = array.new_int(0)
var c_end   = array.new_int(0)
var c_hi    = array.new_float(0)
var c_lo    = array.new_float(0)
var c_avg   = array.new_float(0)

// Entering convergence: lock start, scan back over best_window bars to set bounds
if is_in_conv and not prev_in_conv
    conv_start    := bar_index - best_window + 1
    conv_zone_hi  := high
    conv_zone_lo  := low
    conv_disp_sum := 0.0
    conv_disp_cnt := 0
    d_size_e = array.size(disp_arr)
    for j = 0 to best_window - 1
        // Bounds from price series
        if not na(high[j]) and high[j] > conv_zone_hi
            conv_zone_hi := high[j]
        if not na(low[j]) and low[j] < conv_zone_lo
            conv_zone_lo := low[j]
        // Dispersion sum from buffer
        idx = d_size_e - 1 - j
        if idx >= 0
            v = array.get(disp_arr, idx)
            if not na(v)
                conv_disp_sum += v
                conv_disp_cnt += 1

// Continuing convergence: extend bounds with current bar; accumulate dispersion
if is_in_conv and prev_in_conv
    if not na(high) and high > conv_zone_hi
        conv_zone_hi := high
    if not na(low) and low < conv_zone_lo
        conv_zone_lo := low
    if not na(dispersion)
        conv_disp_sum += dispersion
        conv_disp_cnt += 1

// Exiting convergence: save the zone (single box for the whole period)
if not is_in_conv and prev_in_conv and not na(conv_start)
    duration = bar_index - conv_start
    if duration > 0
        avg_d = conv_disp_cnt > 0 ? conv_disp_sum / conv_disp_cnt : 0.0
        array.push(c_start, conv_start)
        array.push(c_end,   bar_index - 1)
        array.push(c_hi,    conv_zone_hi)
        array.push(c_lo,    conv_zone_lo)
        array.push(c_avg,   avg_d)
        while array.size(c_start) > conv_max_show
            array.shift(c_start)
            array.shift(c_end)
            array.shift(c_hi)
            array.shift(c_lo)
            array.shift(c_avg)
    conv_start    := na
    conv_zone_hi  := na
    conv_zone_lo  := na
    conv_disp_sum := 0.0
    conv_disp_cnt := 0

prev_in_conv := is_in_conv


// ============================================================
// SWING POINT DETECTION (state machine)
// ============================================================

var int   sw_state     = 1
var float sm_extreme   = na
var int   actual_bar   = na
var float actual_price = na
var int   stale        = 0

var swing_all_bars   = array.new_int(0)
var swing_all_prices = array.new_float(0)
var swing_all_types  = array.new_string(0)

sm_current     = sw_state == 1 ? smoothed_high : smoothed_low
actual_current = sw_state == 1 ? high : low

if not na(sm_current)
    improved = na(sm_extreme) ? true :
               (sw_state == 1 ? sm_current > sm_extreme : sm_current < sm_extreme)

    if improved
        sm_extreme   := sm_current
        actual_bar   := bar_index
        actual_price := actual_current
        stale        := 0
    else
        stale += 1
        if not na(actual_price)
            actual_further = sw_state == 1 ? actual_current > actual_price : actual_current < actual_price
            if actual_further
                actual_bar   := bar_index
                actual_price := actual_current

    if stale >= patience and not na(actual_bar)
        conf_type = sw_state == 1 ? "H" : "L"
        array.push(swing_all_bars,   actual_bar)
        array.push(swing_all_prices, actual_price)
        array.push(swing_all_types,  conf_type)
        sw_state     := sw_state * -1
        sm_extreme   := sw_state == 1 ? smoothed_high : smoothed_low
        actual_bar   := bar_index
        actual_price := sw_state == 1 ? high : low
        stale        := 0

if array.size(swing_all_bars) > 60
    array.shift(swing_all_bars)
    array.shift(swing_all_prices)
    array.shift(swing_all_types)

    else

            array.push(f_end,  bar_index - 1)
            array.push(f_hi,   pole_hi)
            array.push(f_lo,   pole_lo)
            array.push(f_cnt,  ib_count)
            total_flags += 1

            while array.size(f_pole) > flag_max_show
                array.shift(f_pole)
                array.shift(f_end)
                array.shift(f_hi)
                array.shift(f_lo)
                array.shift(f_cnt)

        pole_bar := bar_index
        pole_hi  := high
        pole_lo  := low
        ib_count := 0

// MACRO TRENDLINES (functions)

var tl_x      = array.new_float(0)
var tl_y      = array.new_float(0)
var tl_result = array.new_float(2, 0.0)

collect_points(target_type, n_pts) =>
    array.clear(tl_x)
    array.clear(tl_y)
    sz = array.size(swing_all_bars)
    if sz > 0
        for i = 0 to sz - 1
            if array.get(swing_all_types, i) == target_type
                array.push(tl_x, float(array.get(swing_all_bars, i)))
                array.push(tl_y, array.get(swing_all_prices, i))
    tent_type = sw_state == 1 ? "H" : "L"
    if tent_type == target_type and not na(actual_bar) and not na(actual_price)
        array.push(tl_x, float(actual_bar))
        array.push(tl_y, actual_price)
    while array.size(tl_x) > n_pts
        array.shift(tl_x)
        array.shift(tl_y)
    array.size(tl_x)

fit_line() =>
    n = array.size(tl_x)
    slope = 0.0
    intercept = 0.0
    if n >= 2
        sum_x  = array.sum(tl_x)
        sum_y  = array.sum(tl_y)
        mean_x = sum_x / n
        mean_y = sum_y / n
        ss_xx = 0.0
        ss_xy = 0.0
        for i = 0 to n - 1
            dx = array.get(tl_x, i) - mean_x
            dy = array.get(tl_y, i) - mean_y
            ss_xx += dx * dx
            ss_xy += dx * dy
        if ss_xx != 0
            slope := ss_xy / ss_xx
            intercept := mean_y - slope * mean_x
    array.set(tl_result, 0, slope)
    array.set(tl_result, 1, intercept)
    n >= 2
